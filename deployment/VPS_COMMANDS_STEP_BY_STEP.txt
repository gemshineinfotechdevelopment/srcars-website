# ==============================================================================
# 🚀 CAR CARE LUXURY DETAILING - VPS DEPLOYMENT CHEAT SHEET
# Domain: srcars.gemshine.tech | Port: 5030 | Database: MongoDB
# ==============================================================================

# ------------------------------------------------------------------------------
# STEP 1: SSH INTO YOUR VPS
# ------------------------------------------------------------------------------
ssh root@YOUR_VPS_IP


# ------------------------------------------------------------------------------
# STEP 2: NAVIGATE OR CLONE PROJECT TO /var/www/
# ------------------------------------------------------------------------------
cd /var/www
# If cloning from Git:
# git clone <your-git-repo-url> car-care
cd car-care


# ------------------------------------------------------------------------------
# STEP 3: OPTION A - RUN FULL AUTOMATED DEPLOYMENT IN ONE COMMAND
# ------------------------------------------------------------------------------
sudo bash deployment/run_full_deployment.sh


# ------------------------------------------------------------------------------
# STEP 3: OPTION B - RUN STEP-BY-STEP (MANUAL)
# ------------------------------------------------------------------------------

# 3.1 Install MongoDB on VPS:
sudo bash deployment/01_install_mongodb.sh

# 3.2 Build Frontend & Start Backend on Port 5030 with PM2 & Configure Nginx:
sudo bash deployment/02_deploy_app.sh

# 3.3 Activate Free SSL (HTTPS) with Certbot:
sudo bash deployment/03_setup_ssl.sh


# ------------------------------------------------------------------------------
# STEP 4: VERIFY RUNNING SERVICES
# ------------------------------------------------------------------------------

# Check PM2 backend status:
pm2 status

# View live backend logs:
pm2 logs car-care-app

# Check MongoDB service:
sudo systemctl status mongod

# Check Nginx service:
sudo systemctl status nginx

# Test API response:
curl http://127.0.0.1:5030/api/health
curl http://127.0.0.1:5030/api/services


# ------------------------------------------------------------------------------
# STEP 5: USEFUL PM2 MANAGEMENT COMMANDS
# ------------------------------------------------------------------------------

# Restart the application:
pm2 restart car-care-app

# Zero-downtime reload:
pm2 reload car-care-app

# Stop the application:
pm2 stop car-care-app

# Make PM2 restart automatically after VPS reboot:
pm2 startup
pm2 save
